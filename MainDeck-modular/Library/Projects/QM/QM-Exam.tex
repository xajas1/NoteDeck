\begin{enumerate}

%---------------------- 1 ----------------------
\item \textbf{Geladenes Teilchen im Magnetfeld}\\
Ein Teilchen der Masse \(m\) und Ladung \(q\) bewege sich in einem magnetischen Vektorpotential \(\mathbf A(\mathbf r)\).
\begin{enumerate}
  \item Zeigen Sie, dass dieses System (in geeigneten Koordinaten/Eichungen) durch einen harmonischen Oszillator beschrieben werden kann.
  \item Bestimmen Sie die Energieeigenwerte des Systems.
\end{enumerate}

%---------------------- 2 ----------------------
\item \textbf{Komplexes \(\delta\)-Potential}\\
Betrachten Sie das eindimensionale Potential
\[
V(x) = \bigl(\alpha + \mathrm i\,\gamma\bigr)\,\delta(x)\,,
\]
wobei \(\alpha,\gamma\in\mathbb R\).
\begin{enumerate}
  \item Begründen Sie physikalisch, wie der Ansatz der Wellenfunktion für die Streuung an \(x=0\) aussehen muss.
  \item Bestimmen Sie Reflexions- und Transmissionskoeffizient \(R\) und \(T\).
  \item Gilt \(R+T=1\)? Falls nein, warum nicht?
\end{enumerate}

%---------------------- 3 ----------------------
\item \textbf{Zeitentwicklung im unendlichen Potentialkasten}\\
Ein Teilchen befinde sich im unendlichen Potentialkasten mit \(V(x)=0\) für \(0<x<L\) und \(V(x)=\infty\) sonst. Der Anfangszustand sei
\[
\psi(x,0) \;=\; \Bigl[\sin\!\Bigl(\tfrac{2\pi x}{L}\Bigr) + \cos\!\Bigl(\tfrac{3\pi x}{L}\Bigr)\Bigr]\,
\sin\!\Bigl(\tfrac{\pi x}{L}\Bigr).
\]
\begin{enumerate}
  \item Bestimmen Sie die zum Zeitpunkt \(t=0\) sowie zum Zeitpunkt \(t>0\) möglichen messbaren Energien (Eigenwerte) und die zugehörigen Wahrscheinlichkeiten.
  \item Berechnen Sie den Erwartungswert der Energie \(\langle H\rangle(t)\) sowie die Energieunschärfe \(\Delta E(t)\) zum Zeitpunkt \(t\).
  \item Wie müssen die Erwartungswerte \(\langle x\rangle(t)\) und \(\langle p\rangle(t)\) qualitativ aussehen?
\end{enumerate}

%---------------------- 4 ----------------------
\item \textbf{Endlich-dimensionaler Operator}\\
Es sei
\[
H \;=\; \varepsilon\,\mathbb I \;+\; \lambda\,(A + A^\dagger),\qquad
A \;=\; \ket{2}\!\bra{1} + \ket{1}\!\bra{2} + \ket{2}\!\bra{3}.
\]
% Hinweis: So wie definiert ist A nicht unitär (AA^\dagger \neq \mathbb I). Die Teilaufgabe (a) ist in dieser Form widersprüchlich.
\begin{enumerate}
  \item Zeigen Sie, dass \(A\) unitär ist und dass \(A\) mit \(H\) kommutiert.
  \item Bestimmen Sie die Eigenwerte und Eigenvektoren von \(H\).
\end{enumerate}

%---------------------- 5 ----------------------
\item \textbf{Drehimpuls mit Querfeld}\\
Gegeben sei
\[
H \;=\; H_0 + H_1,\qquad
H_0 \;=\; a\,\mathbf L^2 + b\,L_z,\qquad
H_1 \;=\; c\,L_x,
\]
mit Konstanten \(a,b,c\in\mathbb R\) und den üblichen Drehimpulsoperatoren \(\mathbf L^2, L_x, L_z\).
\begin{enumerate}
  \item Bestimmen Sie die Energieeigenwerte und Eigenzustände von \(H_0\).
  \item Berechnen Sie die Störung der Energie zur führenden Ordnung in \(c\).
  \item Definieren Sie einen geeigneten neuen Operator \(J\) und berechnen Sie den exakten Wert der Energiekorrektur.
\end{enumerate}

\end{enumerate}

  \item Zeigen Sie in der Landau-Eichung \(\mathbf A=(0,Bx,0)\), dass \(H\) einem 1D-Oszillator in \(x\) mit Frequenz \(\omega_c=|q|B/m\) entspricht (bis auf freie Bewegung in \(y\)).
  \item Bestimmen Sie die Energieeigenwerte und die Entartung pro Flächeninhalt.
  \item Skizzieren Sie kurz, warum das Ergebnis eichinvariant ist (Vergleich zur symmetrischen Eichung).

\item \textbf{PT-symmetrisches Doppeldelta}\\
Betrachten Sie
\[
V(x)=\alpha\bigl[\delta(x-a)+\delta(x+a)\bigr]+ i\gamma\bigl[\delta(x-a)-\delta(x+a)\bigr],\quad \alpha,\gamma\in\mathbb R.
\]
\begin{enumerate}
  \item Geben Sie den allgemeinen Streuansatz für \(x< -a\), \(-a<x<a\), \(x>a\) an.
  \item Bestimmen Sie die linearen Matching-Bedingungen an \(x=\pm a\).
  \item Diskutieren Sie qualitativ, wann \(R+T\neq1\) wird, und wie sich dies im Wahrscheinlichkeitsstrom äußert.
\end{enumerate}

\item \textbf{Zeitentwicklung im endlichen Kasten \([0,L]\)}\\
Der Anfangszustand sei
\(\displaystyle \psi(x,0)=\mathcal N\,x(L-x)\).
\begin{enumerate}
  \item Bestimmen Sie die Fourier-Koeffizienten \(c_n=\braket{n|\psi(0)}\) der Kasten-Eigenzustände \(\phi_n(x)=\sqrt{\tfrac{2}{L}}\sin(\tfrac{n\pi x}{L})\).
  \item Geben Sie \(\psi(x,t)\), die Wahrscheinlichkeiten \(p_n=|c_n|^2\) und \(\langle H\rangle,\Delta E\) an.
  \item Diskutieren Sie qualitativ den zeitlichen Verlauf von \(\langle x\rangle(t)\) (Revival-Phänomene).
\end{enumerate}

\item \textbf{Dreistufiges System, entartete Störung}\\
Es sei
\[
H=E_0\ket1\!\bra1+E_0\ket2\!\bra2+(E_0+\Delta)\ket3\!\bra3
+\lambda\bigl(\ket1\!\bra2+\ket2\!\bra1\bigr),
\]
mit \(\Delta>0\).
\begin{enumerate}
  \item Diagonalisieren Sie den entarteten Unterraum \(\mathrm{span}\{\ket1,\ket2\}\) und geben Sie Eigenwerte/-zustände von \(H\) an.
  \item Fügen Sie eine schwache Kopplung \(\mu(\ket2\!\bra3+\ket3\!\bra2)\) hinzu. Bestimmen Sie die Energieverschiebungen bis zur führenden nichtverschwindenden Ordnung in \(\mu\) (Rayleigh-Schrödinger).
\end{enumerate}

\item \textbf{Spin-\(\tfrac12\) im schiefen Feld (exakt)}\\
\[
H=\frac{\hbar\Omega}{2}\bigl(\alpha\,\sigma_z+\beta\,\sigma_x\bigr),\quad \alpha^2+\beta^2=1.
\]
\begin{enumerate}
  \item Bestimmen Sie Eigenwerte und normierte Eigenzustände von \(H\).
  \item Für \(\rho(0)=\tfrac12(\mathbf1+m_0\sigma_z)\): Bestimmen Sie \(M^z(t)=\mathrm{Tr}\bigl[\rho(t)\tfrac{\hbar}{2}\sigma_z\bigr]\).
  \item Geben Sie die Resonanzbedingung für maximale Präzession an und interpretieren Sie geometrisch auf der Bloch-Kugel.
\end{enumerate}

\end{enumerate}

  \item Geben Sie die exakten Eigenwerte \(E_{\ell m}=a\,\ell(\ell+1)\hbar^2+\hbar\,m\sqrt{b^2+c^2}\) und die entsprechenden Eigenzustände an.

  \item Leiten Sie die Bewegungsgleichung für \(\langle x\rangle(t)\) her. Diskutieren Sie den Unterschied zur klassischen Bewegungsgleichung.

\item \textbf{Heisenberg-Bild}\\
Betrachten Sie den 1D-HO mit \(H=\hbar\omega\,(a^\dagger a+\tfrac12)\).
\begin{enumerate}
  \item Bestimmen Sie die Zeitentwicklung der Operatoren \(a(t),a^\dagger(t)\) im Heisenberg-Bild.
  \item Berechnen Sie \(\hat x_H(t), \hat p_H(t)\) und den Kommutator \([\hat x_H(t),\hat x_H(t')]\).
  \item Diskutieren Sie kurz, warum dieses Resultat die Unschärferelation reflektiert.
\end{enumerate}

%---------------------- 3 ----------------------
\item \textbf{Variationsprinzip}\\
Ein Teilchen im Potential \(V(x)=\tfrac12 m \omega^2 x^2\) werde mit dem Variationsansatz
\[
\psi_\alpha(x)=\bigl(\tfrac{\alpha}{\pi}\bigr)^{1/4}\exp\!\left(-\tfrac12\alpha x^2\right),\qquad \alpha>0,
\]
approximiert.
\begin{enumerate}
  \item Berechnen Sie den Erwartungswert \(\langle H\rangle(\alpha)\).
  \item Bestimmen Sie das optimale \(\alpha\) und den zugehörigen Energie-Approximation.
  \item Vergleichen Sie das Ergebnis mit der exakten Grundzustandsenergie.
\end{enumerate}

%---------------------- 4 ----------------------
\item \textbf{Dichtematrix und gemischte Zustände}\\
Eine Zweizustandsdichte­matrix sei gegeben durch
\[
\rho=\tfrac12\begin{pmatrix}1 & \eta \\ \eta^* & 1\end{pmatrix},\qquad |\eta|\leq 1.
\]
\begin{enumerate}
  \item Bestimmen Sie die Eigenwerte von \(\rho\).
  \item Unter welcher Bedingung beschreibt \(\rho\) einen reinen Zustand?
  \item Berechnen Sie die Erwartungswerte \(\langle\sigma_x\rangle,\langle\sigma_y\rangle,\langle\sigma_z\rangle\) und interpretieren Sie diese geometrisch auf der Bloch-Kugel.
\end{enumerate}

%---------------------- 5 ----------------------
\item \textbf{Zeitabhängige Störung, Übergangswahrscheinlichkeiten}\\
Ein System mit Energieniveaus \(|0\rangle, |1\rangle\) (Abstand \(\hbar\omega_0\)) wird mit
\[
V(t)=\hbar\Omega\cos(\omega t)\,\bigl(\,|1\rangle\langle 0|+|0\rangle\langle 1|\,\bigr)

gestört.
\begin{enumerate}
  \item Leiten Sie im Rahmen der zeitabhängigen Störungstheorie (1. Ordnung) den Ausdruck für die Übergangswahrscheinlichkeit \(P_{0\to 1}(t)\) her.
  \item Diskutieren Sie die Resonanzbedingung \(\omega\approx\omega_0\) und die Rabi-Oszillationen.
  \item Was passiert im Limes \(\omega\to 0\)?
\end{enumerate}

\end{enumerate}

\end{cases}

\begin{enumerate}

  \[
  \sqrt{\tfrac{2m}{\hbar^2}W - \tfrac{\pi^2}{4a^2}} = \tfrac{2m}{\hbar^2}G

  gilt, und bestimmen Sie die zugehörige Energie.
\end{enumerate}
