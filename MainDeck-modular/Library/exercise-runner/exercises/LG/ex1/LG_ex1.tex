\begin{aligned}
\mathrm{O}(n, \mathbb{C}) & :=\left\{A \in \mathrm{GL}_n(\mathbb{C}): A^{\top} A=I_n\right\} \\
\mathrm{SO}(n, \mathbb{C}) & :=\left\{A \in \mathrm{GL}_n(\mathbb{C}): A^{\top} A=I_n \text { und } \operatorname{det} A=1\right\} .
\end{aligned}
$$
